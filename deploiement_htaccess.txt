# Site statique du Marché Mvog-Mbi (remplace les règles de WordPress)
DirectoryIndex index.html

RewriteEngine On

# Une seule adresse : https://www.marchemvogmbi.com (sans www ou en http -> redirigé)
# .well-known reste libre (renouvellement du certificat). Les sous-domaines ne sont pas touchés.
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTP_HOST} ^marchemvogmbi\.com$ [NC]
RewriteRule ^ https://www.marchemvogmbi.com%{REQUEST_URI} [L,R=301]
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP_HOST} ^www\.marchemvogmbi\.com$ [NC]
RewriteRule ^ https://www.marchemvogmbi.com%{REQUEST_URI} [L,R=301]

# /index.html -> /  et  /en/index.html -> /en/
RewriteCond %{THE_REQUEST} \s/+((?:[a-z]{2}/)?)index\.html[\s?]
RewriteRule ^ /%1 [L,R=301]

# Anciennes adresses WordPress : vers l'accueil
RewriteRule ^(wp-admin|wp-login\.php|xmlrpc\.php)(/.*)?$ / [R=301,L]

# Les données internes ne sont pas listées
Options -Indexes

# Fichier de suivi de la mise en ligne automatique (GitHub Actions) : jamais servi
<Files ".ftp-deploy-sync-state.json">
  Require all denied
</Files>

# Images, styles et scripts gardés en cache par le navigateur
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/jpeg "access plus 30 days"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType text/css "access plus 1 day"
  ExpiresByType application/javascript "access plus 1 day"
  ExpiresByType application/json "access plus 1 hour"
</IfModule>
